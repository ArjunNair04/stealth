# Bind native binaries and wheels to the source files they actually compiled.
# Paths and SHA-256 lines match tools/build_identity.py, including source archives.
file(GLOB_RECURSE stealth_identity_sources CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
    "src/*.cpp" "src/*.hpp" "include/*.hpp" "bindings/*.cpp"
    "python/*.py" "cmake/*.in" "cmake/*.cmake" "app/*.cpp" "bench/*.cpp")
list(APPEND stealth_identity_sources CMakeLists.txt pyproject.toml tools/build_identity.py)
list(SORT stealth_identity_sources)
set(stealth_identity_material "")
foreach(source IN LISTS stealth_identity_sources)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_sha)
    string(APPEND stealth_identity_material "${source}:${source_sha}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
string(SHA256 STEALTH_SOURCE_FINGERPRINT "${stealth_identity_material}")

configure_file(cmake/build_config.hpp.in generated/stealth/build_config.hpp @ONLY)

function(stealth_development_options target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus)
        if(STEALTH_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
        if(STEALTH_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(STEALTH_ENABLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

add_library(stealth_core STATIC src/runtime.cpp src/dense_avx.cpp src/numeric.cpp src/feed_forward.cpp src/attention.cpp src/transformer_block.cpp src/forecaster.cpp src/model_file.cpp src/sha256.cpp src/window_stream.cpp src/signal_index.cpp)
add_library(stealth::core ALIAS stealth_core)
set_target_properties(stealth_core PROPERTIES EXPORT_NAME core POSITION_INDEPENDENT_CODE ON)
target_compile_features(stealth_core PUBLIC cxx_std_20)
target_include_directories(stealth_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_definitions(stealth_core PRIVATE STEALTH_BUILD_CONFIGURATION="$<CONFIG>")
stealth_development_options(stealth_core)
target_compile_definitions(stealth_core PRIVATE STEALTH_ENABLE_AVX=$<BOOL:${STEALTH_ENABLE_AVX}>)
# Preserve explicit accumulation order; never trade finite checks for fast-math.
if(MSVC)
    target_compile_options(stealth_core PRIVATE /fp:precise)
else()
    target_compile_options(stealth_core PRIVATE -ffp-contract=off)
endif()
